module Golden.Game.Agents;
profile Core;
import Standard.Collection.Stores;

enum Intent
{
    Idle,
    Seek(int x, int y),
    Strike(int target)
}

struct Agent
{
    int x;
    int y;
    int health;
    Intent intent;
}

struct World
{
    DenseStore<Agent, 64> agents;
    int tick;
}

Result<Id<Agent>, StoreError> Spawn(ref World world, int x, int y)
{
    Agent agent = Agent{x, y, 100, Intent::Idle};
    return Append<Agent, 64>(ref world.agents, agent);
}

void Tick(ref World world)
{
    for (int i = 0; i < Count<Agent, 64>(ref const world.agents); ++i)
    {
        Id<Agent> id = IdAt<Agent, 64>(ref const world.agents, i)!;
        ref Agent agent = DenseGetMutable<Agent, 64>(ref world.agents, id)!;
        match (agent.intent)
        {
            Intent::Idle => { }
            Intent::Seek(x, y) => {
                if (agent.x < x) { agent.x++; }
                if (agent.y < y) { agent.y++; }
            }
            Intent::Strike(target) => {
                if (target == i) { agent.health -= 1; }
            }
        }
    }
    world.tick++;
}
