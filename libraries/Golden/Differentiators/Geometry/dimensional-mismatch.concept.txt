module Golden.Differentiators.Geometry.DimensionalMismatch;
profile Core;

// Must reject: length-valued matrix times length-valued vector produces m^2,
// not the declared m-valued destination.
float<m> BadContraction()
{
    tensor<float<m>> matrix[1, 1] = [[interpret 2.0 as float<m>]];
    tensor<float<m>> vector[1] = [interpret 3.0 as float<m>];
    tensor<float<m>> output[1] = interpret 0.0 as float<m>;
    output[i] = matrix[i, j] * vector[j];
    return output[0];
}
